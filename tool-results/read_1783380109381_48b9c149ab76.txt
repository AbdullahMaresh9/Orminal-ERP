    95→function lineTotal(line: { quantity: number; unitPrice: number; discount: number; taxRate: number }): number {
    96→  const qty = Number(line.quantity) || 0
    97→  const price = Number(line.unitPrice) || 0
    98→  const disc = Number(line.discount) || 0
    99→  const tax = Number(line.taxRate) || 0
   100→  const net = Math.max(0, qty * price - disc)
   101→  return net + net * (tax / 100)
   102→}
