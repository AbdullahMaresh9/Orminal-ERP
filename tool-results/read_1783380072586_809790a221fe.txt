   106→   106→function lineTotal(line: { quantity: number; unitPrice: number; discount: number; taxRate: number }): number {
   107→   107→  const qty = Number(line.quantity) || 0
   108→   108→  const price = Number(line.unitPrice) || 0
   109→   109→  const disc = Number(line.discount) || 0
   110→   110→  const tax = Number(line.taxRate) || 0
   111→   111→  const net = qty * price - disc
   112→   112→  return Math.max(0, net) + Math.max(0, net) * (tax / 100)
   113→   113→}
