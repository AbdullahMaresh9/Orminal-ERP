   438→// Add Journal Dialog — with dynamic line editor + balance validation
   439→// ====================================================================
   440→function AddJournalDialog({
   441→  open, onOpenChange, accounts,
   442→}: {
   443→  open: boolean
   444→  onOpenChange: (o: boolean) => void
   445→  accounts: Account[]
   446→}) {
   447→  const qc = useQueryClient()
   448→  const [date, setDate] = useState<string>(new Date().toISOString().slice(0, 10))
   449→  const [description, setDescription] = useState('')
   450→  const [refType, setRefType] = useState('manual')
   451→  const [lines, setLines] = useState<JournalLine[]>([
   452→    { accountId: '', accountCode: '', debit: 0, credit: 0, description: '' },
   453→    { accountId: '', accountCode: '', debit: 0, credit: 0, description: '' },
   454→  ])
   455→
   456→  const totals = useMemo(() => {
   457→    const debit = lines.reduce((s, l) => s + (Number(l.debit) || 0), 0)
   458→    const credit = lines.reduce((s, l) => s + (Number(l.credit) || 0), 0)
   459→    return { debit, credit }
   460→  }, [lines])
   461→
   462→  const balanced = Math.abs(totals.debit - totals.credit) < 0.01
   463→  const hasZeroRows = lines.some((l) => !l.accountCode)
   464→  const hasEmptyAmounts = lines.some((l) => l.debit === 0 && l.credit === 0)
   465→
   466→  function updateLine(idx: number, patch: Partial<JournalLine>) {
   467→    setLines((prev) => prev.map((l, i) => i === idx ? { ...l, ...patch } : l))
   468→  }
   469→  function updateAccount(idx: number, code: string) {
   470→    const acc = accounts.find((a) => a.code === code)
   471→    setLines((prev) => prev.map((l, i) => i === idx ? {
   472→      ...l,
   473→      accountCode: code,
   474→      accountId: acc?.id ?? '',
   475→      account: acc,
   476→    } : l))
   477→  }
   478→  // Enter debit → clear credit, vice versa
   479→  function setDebit(idx: number, val: string) {
   480→    const n = Number(val) || 0
   481→    updateLine(idx, { debit: n, credit: 0 })
   482→  }
   483→  function setCredit(idx: number, val: string) {
   484→    const n = Number(val) || 0
   485→    updateLine(idx, { credit: n, debit: 0 })
   486→  }
   487→  function addRow() {
   488→    setLines((prev) => [...prev, { accountId: '', accountCode: '', debit: 0, credit: 0, description: '' }])
   489→  }
   490→  function removeRow(idx: number) {
   491→    setLines((prev) => prev.filter((_, i) => i !== idx))
   492→  }
   493→
   494→  const saveMutation = useMutation({
   495→    mutationFn: async () => {
   496→      const payload = {
   497→        date,
   498→        description,
   499→        refType,
   500→        status: 'posted',
   501→        lines: lines.map((l) => ({
   502→          accountCode: l.accountCode,
   503→          debit: Number(l.debit) || 0,
   504→          credit: Number(l.credit) || 0,
   505→          description: l.description,
   506→        })),
   507→      }
